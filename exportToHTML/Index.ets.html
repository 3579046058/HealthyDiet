@Entry
@Component
struct Index {
  @State showTime: boolean = false;
  controller: TextClockController = new TextClockController();

  build() {
    Row() {
      Column() {
        if (this.showTime) {
          TextClock()
            .format('HH:mm:ss')
            .fontSize(40)
            .fontWeight(FontWeight.Bold)
            .margin({ bottom: 30 })
        }

        Button('Hello World')
          .fontSize($r('app.float.page_text_font_size'))
          .fontWeight(FontWeight.Bold)
          .onClick(() => {
            this.showTime = true;
          })
      }
      .width('100%')
      .justifyContent(FlexAlign.Center)
    }
    .height('100%')
  }
}
